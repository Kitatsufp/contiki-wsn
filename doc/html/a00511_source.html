205 dtls_handle_read(struct dtls_context_t *ctx) {
206  int fd;

213 
214  if (!fd)
215  return -1;
216 
217  memset(&session, 0, sizeof(session_t));

219  len = recvfrom(fd, buf, MAX_READ_BUF, 0,

221

228  }
229 
230  return dtls_handle_message(ctx, &session, buf, len);

336  struct timeval timeout;
337  unsigned short port = DEFAULT_PORT;

345  dtls_init();

348 #ifdef DTLS_PSK
349  psk_id_length = strlen(PSK_DEFAULT_IDENTITY);

353 #endif /* DTLS_PSK */

355  while ((opt = getopt(argc, argv, "p:o:v:" PSK_OPTIONS)) != -1) {
356  switch (opt) {
357 #ifdef DTLS_PSK
358  case 'i' : {
359  ssize_t result = read_from_file(optarg, psk_id, PSK_ID_MAXLEN);
360  if (result < 0) {
361  dtls_warn("cannot read PSK identity\n");
362  } else {
363  psk_id_length = result;
364  }
365  break;
366  }
367  case 'k' : {
368  ssize_t result = read_from_file(optarg, psk_key, PSK_MAXLEN);
369  if (result < 0) {
370  dtls_warn("cannot read PSK\n");
371  } else {
372  psk_key_length = result;
373  }
374  break;
375  }
376 #endif /* DTLS_PSK */
377  case 'p' :
378  strncpy(port_str, optarg, NI_MAXSERV-1);
379  port_str[NI_MAXSERV - 1] = '\0';
380  break;
381  case 'o' :
382  output_file.length = strlen(optarg);
383  output_file.s = (unsigned char *)malloc(output_file.length + 1);
384 
385  if (!output_file.s) {
386  dtls_crit("cannot set output file: insufficient memory\n");
387  exit(-1);
388  } else {
389  /* copy filename including trailing zero */
390  memcpy(output_file.s, optarg, output_file.length + 1);
391  }
392  break;
393  case 'v' :
394  log_level = strtol(optarg, NULL, 10);
395  break;
396  default:
397  usage(argv[0], dtls_package_version());
398  exit(1);
399  }
400  }
401 
402  dtls_set_log_level(log_level);
403 
404  if (argc <= optind) {
405  usage(argv[0], dtls_package_version());
406  exit(1);
407  }
408 
409  memset(&dst, 0, sizeof(session_t));
410  /* resolve destination address where server should be sent */
411  res = resolve_address(argv[optind++], &dst.addr.sa);
412  if (res < 0) {
413  dtls_emerg("failed to resolve address\n");
414  exit(-1);
415  }
416  dst.size = res;
417 
418  /* use port number from command line when specified or the listen
419  port, otherwise */
420  dst.addr.sin.sin_port = htons(atoi(optind < argc ? argv[optind++] : port_str));
421 
422 
423  /* init socket and set it to non-blocking */
424  fd = socket(dst.addr.sa.sa_family, SOCK_DGRAM, 0);
425 
426  if (fd < 0) {
427  dtls_alert("socket: %s\n", strerror(errno));
428  return 0;
429  }
430 
431  if (setsockopt(fd, SOL_SOCKET, SO_REUSEADDR, &on, sizeof(on) ) < 0) {
432  dtls_alert("setsockopt SO_REUSEADDR: %s\n", strerror(errno));
433  }
434 #if 0
435  flags = fcntl(fd, F_GETFL, 0);
436  if (flags < 0 || fcntl(fd, F_SETFL, flags | O_NONBLOCK) < 0) {
437  dtls_alert("fcntl: %s\n", strerror(errno));
438  goto error;
439  }
440 #endif
441  on = 1;
442 #ifdef IPV6_RECVPKTINFO
443  if (setsockopt(fd, IPPROTO_IPV6, IPV6_RECVPKTINFO, &on, sizeof(on) ) < 0) {
444 #else /* IPV6_RECVPKTINFO */
445  if (setsockopt(fd, IPPROTO_IPV6, IPV6_PKTINFO, &on, sizeof(on) ) < 0) {
446 #endif /* IPV6_RECVPKTINFO */
447  dtls_alert("setsockopt IPV6_PKTINFO: %s\n", strerror(errno));
448  }
449 
450  if (signal(SIGINT, dtls_handle_signal) == SIG_ERR) {
451  dtls_alert("An error occurred while setting a signal handler.\n");
452  return EXIT_FAILURE;
453  }
454 
455  dtls_context = dtls_new_context(&fd);
456  if (!dtls_context) {
457  dtls_emerg("cannot create context\n");
458  exit(-1);
459  }
460

462 
463  dtls_connect(dtls_context, &dst);
464 
465  while (1) {
466  FD_ZERO(&rfds);
467  FD_ZERO(&wfds);
468 
469  FD_SET(fileno(stdin), &rfds);
470  FD_SET(fd, &rfds);

472

474  timeout.tv_usec = 0;
475

477 
478  if (result < 0) { /* error */
479  if (errno != EINTR)
480  perror("select");
481  } else if (result == 0) { /* timeout */
482  } else { /* ok */
483  if (FD_ISSET(fd, &wfds))
484  /* FIXME */;
485  else if (FD_ISSET(fd, &rfds))
486  dtls_handle_read(dtls_context);
487  else if (FD_ISSET(fileno(stdin), &rfds))
488  handle_stdin();
489  }
490 
491  if (len) {
492  if (len >= strlen(DTLS_CLIENT_CMD_CLOSE) &&
493  !memcmp(buf, DTLS_CLIENT_CMD_CLOSE, strlen(DTLS_CLIENT_CMD_CLOSE))) {
494  printf("client: closing connection\n");
495  dtls_close(dtls_context, &dst);
496  len = 0;
497  } else if (len >= strlen(DTLS_CLIENT_CMD_RENEGOTIATE) &&
498  !memcmp(buf, DTLS_CLIENT_CMD_RENEGOTIATE, strlen(DTLS_CLIENT_CMD_RENEGOTIATE))) {
499  printf("client: renegotiate connection\n");
500  dtls_renegotiate(dtls_context, &dst);
501  len = 0;
502  } else if (len >= strlen(DTLS_CLIENT_CMD_REHANDSHAKE) &&
503  !memcmp(buf, DTLS_CLIENT_CMD_REHANDSHAKE, strlen(DTLS_CLIENT_CMD_REHANDSHAKE))) {
504  printf("client: rehandshake connection\n");
505  if (orig_dtls_context == NULL) {
506  /* Cache the current context. We cannot free the current context as it will notify
507  * the Server to close the connection (which we do not want).
508  */
509  orig_dtls_context = dtls_context;
510  /* Now, Create a new context and attempt to initiate a handshake. */
511  dtls_context = dtls_new_context(&fd);
512  if (!dtls_context) {
513  dtls_emerg("cannot create context\n");
514  exit(-1);
515  }
516  dtls_set_handler(dtls_context, &cb);
517  dtls_connect(dtls_context, &dst);
518  }
519  len = 0;
520  } else {
521  try_send(dtls_context, &dst);
522  }
523  }
524  }
525 
526  dtls_free_context(dtls_context);
527  dtls_free_context(orig_dtls_context);
528  exit(0);
529 }
