93 dtls_handshake_parameters_t *dtls_handshake_new()
94 {
95  dtls_handshake_parameters_t *handshake;
96 
97  handshake = dtls_handshake_malloc();
98  if (!handshake) {
99  dtls_crit("can not allocate a handshake struct\n");
100  return NULL;
101  }
102 
103  memset(handshake, 0, sizeof(*handshake));
104 
105  if (handshake) {
106  /* initialize the handshake hash wrt. the hard-coded DTLS version */
107  dtls_debug("DTLSv12: initialize HASH_SHA256\n");
108  /* TLS 1.2: PRF(secret, label, seed) = P_<hash>(secret, label + seed) */
109  /* FIXME: we use the default SHA256 here, might need to support other
110  hash functions as well */
111  dtls_hash_init(&handshake->hs_state.hs_hash);
112  }
113  return handshake;
114 }

152 size_t
153 dtls_p_hash(dtls_hashfunc_t h,
154  const unsigned char *key, size_t keylen,
155  const unsigned char *label, size_t labellen,
156  const unsigned char *random1, size_t random1len,
157  const unsigned char *random2, size_t random2len,
158  unsigned char *buf, size_t buflen) {
159  dtls_hmac_context_t *hmac_a, *hmac_p;
160 
161  unsigned char A[DTLS_HMAC_DIGEST_SIZE];
162  unsigned char tmp[DTLS_HMAC_DIGEST_SIZE];
163  size_t dlen; /* digest length */
164  size_t len = 0; /* result length */
165 
166  hmac_a = dtls_hmac_new(key, keylen);
167  if (!hmac_a)
168  return 0;
169 
170  /* calculate A(1) from A(0) == seed */
171  HMAC_UPDATE_SEED(hmac_a, label, labellen);
172  HMAC_UPDATE_SEED(hmac_a, random1, random1len);
173  HMAC_UPDATE_SEED(hmac_a, random2, random2len);
174 
175  dlen = dtls_hmac_finalize(hmac_a, A);
176 
177  hmac_p = dtls_hmac_new(key, keylen);
178  if (!hmac_p)
179  goto error;
180 
181  while (len + dlen < buflen) {
182 
183  /* FIXME: rewrite loop to avoid superflous call to dtls_hmac_init() */
184  dtls_hmac_init(hmac_p, key, keylen);
185  dtls_hmac_update(hmac_p, A, dlen);
186 
187  HMAC_UPDATE_SEED(hmac_p, label, labellen);
188  HMAC_UPDATE_SEED(hmac_p, random1, random1len);
189  HMAC_UPDATE_SEED(hmac_p, random2, random2len);
190 
191  len += dtls_hmac_finalize(hmac_p, tmp);
192  memcpy(buf, tmp, dlen);
193  buf += dlen;
194 
195  /* calculate A(i+1) */
196  dtls_hmac_init(hmac_a, key, keylen);
197  dtls_hmac_update(hmac_a, A, dlen);
198  dtls_hmac_finalize(hmac_a, A);
199  }
200 
201  dtls_hmac_init(hmac_p, key, keylen);
202  dtls_hmac_update(hmac_p, A, dlen);
203 
204  HMAC_UPDATE_SEED(hmac_p, label, labellen);
205  HMAC_UPDATE_SEED(hmac_p, random1, random1len);
206  HMAC_UPDATE_SEED(hmac_p, random2, random2len);
207 
208  dtls_hmac_finalize(hmac_p, tmp);
209  memcpy(buf, tmp, buflen - len);
210 
211  error:
212  dtls_hmac_free(hmac_a);
213  dtls_hmac_free(hmac_p);
214 
215  return buflen;
216 }

332 int dtls_ec_key_from_uint32_asn1(const uint32_t *key, size_t key_size,
333  unsigned char *buf) {
334  int i;
335  unsigned char *buf_orig = buf;
336  int first = 1;
337 
338  for (i = (key_size / sizeof(uint32_t)) - 1; i >= 0 ; i--) {
339  if (key[i] == 0)
340  continue;
341  /* the first bit has to be set to zero, to indicate a poritive integer */
342  if (first && key[i] & 0x80000000) {
343  *buf = 0;
344  buf++;
345  dtls_int_to_uint32(buf, key[i]);
346  buf += 4;
347  } else if (first && !(key[i] & 0xFF800000)) {
348  buf[0] = (key[i] >> 16) & 0xff;
349  buf[1] = (key[i] >> 8) & 0xff;
350  buf[2] = key[i] & 0xff;
351  buf += 3;
352  } else if (first && !(key[i] & 0xFFFF8000)) {
353  buf[0] = (key[i] >> 8) & 0xff;
354  buf[1] = key[i] & 0xff;
355  buf += 2;
356  } else if (first && !(key[i] & 0xFFFFFF80)) {
357  buf[0] = key[i] & 0xff;
358  buf += 1;
359  } else {
360  dtls_int_to_uint32(buf, key[i]);
361  buf += 4;
362  }
363  first = 0;
364  }
365  return buf - buf_orig;
366 }
