51 get_extended_address(const linkaddr_t *addr)
52 #if LINKADDR_SIZE == 2

54  /* workaround for short addresses: derive EUI64 as in RFC 6282 */
55  static linkaddr_extended_t template = { { 0x00 , 0x00 , 0x00 ,

68 void
69 ccm_star_packetbuf_set_nonce(uint8_t *nonce, int forward)
70 {
71  const linkaddr_t *source_addr;
72 
73  source_addr = forward ? &linkaddr_node_addr : packetbuf_addr(PACKETBUF_ADDR_SENDER);
74  memcpy(nonce, get_extended_address(source_addr), 8);
75  nonce[8] = packetbuf_attr(PACKETBUF_ATTR_FRAME_COUNTER_BYTES_2_3) >> 8;
76  nonce[9] = packetbuf_attr(PACKETBUF_ATTR_FRAME_COUNTER_BYTES_2_3) & 0xff;
77  nonce[10] = packetbuf_attr(PACKETBUF_ATTR_FRAME_COUNTER_BYTES_0_1) >> 8;
78  nonce[11] = packetbuf_attr(PACKETBUF_ATTR_FRAME_COUNTER_BYTES_0_1) & 0xff;
79  nonce[12] = packetbuf_attr(PACKETBUF_ATTR_SECURITY_LEVEL);
80 }
