32 /**
33  * \addtogroup apps
34  * @{
35  *
36  * \defgroup ubidots-engine A client for the Ubidots cloud service
37  * @{
38  *
39  * This engine allows a device to connect to the Ubidots cloud service and
40  * push data.
41  *
42  * The engine supports API version defined by #UBIDOTS_API_VER
43  *
44  * To keep code footprint / RAM requirements low, the engine only supports
45  * a single connection with Ubidots and many of the connection's parameters
46  * are selected at compile time. As a consequence of this design decision,
47  * each Contiki image may only have one process using this engine. The rest
48  * of the engine's documentation refers to this process as the 'consumer'
49  * process.
50  *
51  * The engine supports writing multiple data points using a single request:
52  * See the "Collections" section of the Ubidots API docs.
53  *
54  * The engine assumes that the user
55  * - Has previously registered an account and generated a short auth token
56  * - Has created the necessary data sources and variables by using the Ubidots
57  * web interface
58  *
59  * The user MUST satisfy both requirements above before attempting to use the
60  * engine.
61  *
62  * This library's compilation will fail if the auth token has not been defined.
63  *
64  * #define UBIDOTS_CONF_AUTH_TOKEN "<the-token>"
65  *
66  * Include the opening and closing double-quotes.
67  *
68  * A normal workflow would be as follows:
69  * - The application fires up the engine by placing a call to ubidots_init()
70  * If the client is interested in some of the reply's HTTP headers, it can
71  * specify them at this stage.
72  * - Upon reception of the ubidots_event_established event, the application
73  * prepares a POST message by making a single call to ubidots_prepare_post(),
74  * following by one or more calls to ubidots_enqueue_value().
75  * - The application triggers the POST message by calling ubidots_post()
76  * - When the POST has been sent, the engine will generate
77  * ubidots_event_post_sent.
78  * - As soon as the reply starts streaming, the library will generate a series
79  * of synchronous ubidots_event_post_reply_received events. Each event will
80  * pass to the consumer a part of the reply (HTTP status, HTTP headers,
81  * payload). The application will have to process it immediately, or copy it
82  * to separate RAM storage, since the memory will be overwritten as soon as
83  * the application has finished processing the event.
84  *
85  * The library will refuse to send a POST if a previous POST is in-flight.
86  *
87  * The resource to be updated must be specified in calls to
88  * ubidots_enqueue_value() and ubidots_prepare_post() through the \e variable
89  * argument.
90  *
91  * The optional 'timestamp' argument is not supported, due to it being
92  * specified as msec since the epoch (POSIX), which is something not currently
93  * supported by Contiki.
94  *
95  * For more information, on the Ubidots API, visit:
96  * http://ubidots.com/docs/index.html
97  *
98  * Many thanks to Antonio Liñán Colina ([Zolertia](http://zolertia.io)) for
99  * his ideas and contributions, and also to the Ubidots team for the support.
100  */

217 /**
218  * \brief Prepare an HTTP POST request to be used to send data
219  * \param variable A string containing the variable to POST or NULL
220  * \return UBIDOTS_SUCCESS or UBIDOTS_ERROR
221  *
222  * This function will prepare HTTP headers and will otherwise prepare the
223  * engine to accept values.
224  *
225  * This function MUST be called each time the consumer wishes to post data.
226  * This call MUST then be followed by one ore more calls to
227  * ubidots_enqueue_value(), followed by a single call to ubidots_post()
228  *
229  * If the argument \e variable is NULL, then the engine will prepare to post
230  * multiple values with different variables, by POST to this URL:
231  * /api/v1.6/collections/values
232  * In this case, ALL subsequent calls to ubidots_enqueue_value() MUST specify
233  * a non-NULL value for the variable argument
234  *
235  * If the argument \e variable is NOT NULL, then the engine will prepare to
236  * post one ore more values for the same variable, by posting to this URL:
237  * /api/v1.6/variables/<variable>/values
238  * where <variable> is replaced by the value of \e variable
239  * By using this format, all \e value arguments in subsequent calls to
240  * ubidots_enqueue_value() will be with reference to this same variable.
241  * If \e variable is specified here, then ALL subsequent calls to
242  * ubidots_enqueue_value() must specify NULL for the \e variable argument.
243  * Failing to do so will result in HTTP POSTs that violate the Ubidots API
244  */
245 int ubidots_prepare_post(const char *variable);
246 
247 /**
248  * \brief Enqueues a data point to a previously initialised request
249  * \param variable A string containing the variable to POST or NULL
250  * \param value The value to be POSTEd
251  * \return UBIDOTS_SUCCESS or UBIDOTS_ERROR
252  *
253  * If \e value should be encoded as a JSON string, the surrounding double quotes
254  * must be included in a call to this function. Skip them to encode as a JSON
255  * number. To encode as JSON boolean, pass "true" or "false". Some examples:
256  *
257  * - ubidots_enqueue_value(NULL, "32"); will produce {"value":32}
258  * - ubidots_enqueue_value(NULL, "\"32\\""); will produce {"value":"32"}
259  * - ubidots_enqueue_value(NULL, "false"); will produce {"value":false}
260  *
261  * Regarding \e variable, it depends on whether it was specified or not in the
262  * previous call to ubidots_prepare_post(). Read this function's documentation
263  * for more details.
264  */
265 int ubidots_enqueue_value(const char *variable, const char *value);
266 
267 /**
268  * \brief Post previously enqueued data
269  * \return UBIDOTS_SUCCESS or UBIDOTS_ERROR
270  *
271  * This function MUST be preceded by a call to ubidots_prepare_post(),
272  * followed by one or more calls to ubidots_enqueue_value(). The entire HTTP
273  * POST request must fit in a single buffer.
274  *
275  * The caller does NOT need to allocate the buffer, but must make sure that
276  * the outgoing buffer is large enough for the POST request by specifying large
277  * enough a value for UBIDOTS_OUT_BUFFER_SIZE
278  *
279  * The function will return UBIDOTS_ERROR if a previous POST is still in
280  * progress
281  */
282 int ubidots_post(void);
