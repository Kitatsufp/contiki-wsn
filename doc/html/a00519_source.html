349 static int
350 enqueue_chunk(const char *format, ...)
351 {
352  int tmp_buf_len;
353  char tmp_buf[TMP_BUF_SIZE];
354  va_list ap;
355 
356  va_start(ap, format);
357 
358  tmp_buf_len = vsnprintf(tmp_buf, TMP_BUF_SIZE, format, ap);
359 
360  if(tmp_buf_len < 0 ||
361  tmp_buf_len >= UBIDOTS_OUT_BUFFER_SIZE - conn.out_buf_pos - 1) {
362  return UBIDOTS_ERROR;
363  }
364 
365  memcpy(&conn.out_buf[conn.out_buf_pos], tmp_buf, tmp_buf_len);
366  conn.out_buf_pos += tmp_buf_len;
367 
368  return UBIDOTS_SUCCESS;
369 }

383 /*
384  * Data input callback. Here, we process the reply from the Ubidots server.
385  *
386  * We parse the reply as it arrives, notifying the sockets library that we
387  * consumed all input each time. We store the HTTP return code in a separate
388  * field, HTTP reply headers in a MEMB/LIST structure, and the payload as-is in
389  * a buffer.
390  *
391  * Ubidots will send us Transfer-Encoding: chunked. When we have parsed the
392  * final chunk, we generate an event so that the consumer process can see
393  * the reply.
394  */
395 static int
396 data_callback(struct tcp_socket *s, void *ptr, const uint8_t *input_data_ptr,
397  int input_data_len)
398 {
399  ubidots_reply_part_t reply;
400  long int http_status;
401  const char **hdr;
402  const char *buf_ptr = (const char *)input_data_ptr;
403 
404  if(strncmp((const char *)input_data_ptr, "HTTP/",
405  MIN(5, input_data_len)) == 0) {
406 
407  /* Reset the reply parser */
408  conn.reply_parser_state = REPLY_PARSER_STATE_READING_STATUS;
409  }
410 
411  while(buf_ptr < (const char *)(&input_data_ptr[input_data_len])) {
412  switch(conn.reply_parser_state) {
413  case REPLY_PARSER_STATE_READING_STATUS:
414  /*
415  * First chunk. Initialise the reply buffer and then retrieve the HTTP
416  * status code
417  */
418  memset(&conn.reply_payload, 0, UBIDOTS_REPLY_PAYLOAD_MAX_LEN);
419 
420  conn.reply_counter = 0;
421 
422  /*
423  * The first message will always start with something like:
424  * HTTP/1.x 200 OK
425  *
426  * Status always starts at the 10th char
427  */
428  http_status = strtol((char *)(&input_data_ptr[9]), NULL, 10);
429  reply.type = UBIDOTS_REPLY_TYPE_HTTP_STATUS;
430  reply.content = &http_status;
431 
432  PRINTF(UBIDOTS_DEBUG "HTTP Reply %lu\n", http_status);
433 
434  notify_consumer(&reply);
435 
436  /* Sink the remainder of the status line */
437  buf_ptr = (const char *)(&input_data_ptr[9]);
438  while(*buf_ptr != '\n') {
439  buf_ptr++;
440  }
441  buf_ptr++;
442  conn.reply_parser_state = REPLY_PARSER_STATE_NEW_HEADER;
443  break;
444  case REPLY_PARSER_STATE_NEW_HEADER:
445  /*
446  * First, check that this is really a header and not the start of the
447  * payload. If it's the start of the payload, bail out and prepare to
448  * read a chunk.
449  *
450  * We actually have to check both chars and increment by one each time,
451  * otherwise we risk failing if the data in the socket's input buffer
452  * ends after the \r
453  *
454  * We change state when we have encountered \n
455  */
456  if(*buf_ptr == '\r' || *buf_ptr == '\n') {
457  if(*buf_ptr == '\n') {
458  conn.reply_parser_state = REPLY_PARSER_STATE_READING_CHUNK_LEN;
459  memset(conn.reply_tmp_buf, 0, sizeof(conn.reply_tmp_buf));
460  memset(conn.reply_payload, 0, UBIDOTS_REPLY_PAYLOAD_MAX_LEN);
461  conn.reply_tmp_buf_counter = 0;
462  }
463  buf_ptr++;
464  break;
465  }
466 
467  PRINTF(UBIDOTS_DEBUG "New header: <");
468 
469  /*
470  * Zero-out the buffer. We subsequently make sure to leave at least one
471  * free byte, so that takes care of null-termination
472  */
473  memset(conn.reply_payload, 0, UBIDOTS_REPLY_PAYLOAD_MAX_LEN);
474 
475  /* And prepare to populate it */
476  conn.reply_counter = 0;
477  conn.reply_parser_state = REPLY_PARSER_STATE_READING_HEADER;
478  break;
479  case REPLY_PARSER_STATE_READING_HEADER:
480  if(*buf_ptr != '\r' && *buf_ptr != '\n') {
481  /* Truncate if length exceeded */
482  PRINTF("%c", *buf_ptr);
483  if(conn.reply_counter < UBIDOTS_REPLY_PAYLOAD_MAX_LEN - 1) {
484  conn.reply_payload[conn.reply_counter] = *buf_ptr;
485  conn.reply_counter++;
486  }
487  } else {
488  if(*buf_ptr == '\n') {
489  PRINTF(">\n");
490  conn.reply_counter = 0;
491  conn.reply_parser_state = REPLY_PARSER_STATE_NEW_HEADER;
492 
493  hdr = conn.headers;
494  while(*hdr) {
495  if(strncmp(*hdr, conn.reply_payload, strlen(*hdr)) == 0) {
496  PRINTF(UBIDOTS_DEBUG "Client wants header '%s'\n", *hdr);
497  reply.type = UBIDOTS_REPLY_TYPE_HTTP_HEADER;
498  reply.content = conn.reply_payload;
499  notify_consumer(&reply);
500  break;
501  }
502  hdr++;
503  }
504  }
505  }
506  buf_ptr++;
507  break;
508  case REPLY_PARSER_STATE_READING_CHUNK_LEN:
509  /* Discard \r, read chunk length after \n */
510  if(*buf_ptr != '\r' && *buf_ptr != '\n') {
511  conn.reply_tmp_buf[conn.reply_tmp_buf_counter] = *buf_ptr;
512  conn.reply_tmp_buf_counter++;
513  } else {
514  if(*buf_ptr == '\n') {
515  conn.reply_chunk_len = 0;
516  /* Read out hex length for this chunk */
517  conn.reply_parser_state = REPLY_PARSER_STATE_READING_PAYLOAD;
518  conn.reply_chunk_len = strtol(conn.reply_tmp_buf, NULL, 16);
519 
520  PRINTF(UBIDOTS_DEBUG "Chunk, len %d: <", conn.reply_chunk_len);
521 
522  /* Wipe temporary buffer for next pass */
523  memset(conn.reply_tmp_buf, 0, sizeof(conn.reply_tmp_buf));
524 
525  /* Check for last chunk */
526  if(conn.reply_chunk_len == 0) {
527  conn.reply_parser_state = REPLY_PARSER_STATE_DONE;
528  PRINTF("(End of Reply)> (Payload Length %u bytes)\n",
529  conn.reply_counter);
530 
531  reply.type = UBIDOTS_REPLY_TYPE_PAYLOAD;
532  reply.content = conn.reply_payload;
533  notify_consumer(&reply);
534  }
535  }
536  }
537  buf_ptr++;
538  break;
539  case REPLY_PARSER_STATE_READING_PAYLOAD:
540  if(conn.reply_chunk_len > 0 &&
541  conn.reply_counter < UBIDOTS_REPLY_PAYLOAD_MAX_LEN - 1) {
542  PRINTF("%c", *buf_ptr);
543  conn.reply_payload[conn.reply_counter] = *buf_ptr;
544  conn.reply_counter++;
545  conn.reply_chunk_len--;
546  } else {
547  /* Done with this chunk */
548  PRINTF("> (counter = %u)\n", conn.reply_counter);
549  conn.reply_parser_state = REPLY_PARSER_STATE_READING_CHUNK_LEN;
550  }
551  buf_ptr++;
552  break;
553  case REPLY_PARSER_STATE_DONE:
554  case REPLY_PARSER_STATE_ERROR:
555  default:
556  /*
557  * Sink the remainder if we reach here due to some error or because
558  * we're done parsing
559  */
560  buf_ptr++;
561  break;
562  }
563  }
564 
565  return 0;
566 }

601 static int
602 http_prepare_header(const char *variable)
603 {
604  /* If variable == NULL, we are sending a collection */
605  if(variable == NULL) {
606  if(enqueue_chunk(HEADER_HTTP_REQUEST_C) == UBIDOTS_ERROR) {
607  return UBIDOTS_ERROR;
608  }
609  } else {
610  if(enqueue_chunk(HEADER_HTTP_REQUEST_V, variable) == UBIDOTS_ERROR) {
611  return UBIDOTS_ERROR;
612  }
613  }
614 
615  /* Send Host: if host_header has a valid string */
616  if(conn.host_header[0] != 0) {
617  if(enqueue_chunk(HEADER_HOST, conn.host_header) == UBIDOTS_ERROR) {
618  return UBIDOTS_ERROR;
619  }
620  }
621 
622  if(enqueue_chunk(HEADER_CONTENT_TYPE) == UBIDOTS_ERROR) {
623  return UBIDOTS_ERROR;
624  }
625 
626  /*
627  * Content-Length: Business as usual
628  * We have no idea how much it'll be... We can either put the entire
629  * payload in a separate buffer, count it, then create the HTTP POST
630  * Or we can skip the secondary buffer and leave Content-Length empty to
631  * start with, until just before we send out the POST. We go for the latter
632  * here to save RAM.
633  *
634  * We do know that the actual value will be written at the current buffer
635  * position, plus the length of 'Content-Length: ' (16)
636  * We also know that it can't really be longer than say 5 bytes. So we
637  * write 5 spaces for now and we adjust later with the actual value.
638  */
639  conn.content_length_pos = conn.out_buf_pos + HEADER_CONTENT_LEN_LEN;
640  if(enqueue_chunk(HEADER_CONTENT_LENGTH) == UBIDOTS_ERROR) {
641  return UBIDOTS_ERROR;
642  }
643 
644  if(enqueue_chunk(HEADER_X_AUTH_TOKEN "\r\n") == UBIDOTS_ERROR) {
645  return UBIDOTS_ERROR;
646  }
647 
648  return UBIDOTS_SUCCESS;
649 }

651 /* Encode the hostname to be used as the value for the HTTP Host: header */
652 static void
653 encode_http_host(uint8_t host_is_ip_address)
654 {
655  int remaining = UBIDOTS_HOST_HEADER_LEN;
656  int host_len;
657 
658  if(host_is_ip_address && NETSTACK_CONF_WITH_IPV6) {
659  host_len = snprintf(conn.host_header, remaining,
660  "[%s]", UBIDOTS_REMOTE_HOST);
661  } else {
662  host_len = snprintf(conn.host_header, remaining,
663  "%s", UBIDOTS_REMOTE_HOST);
664  }
665 
666  if(host_len < 0 || host_len >= remaining) {
667  printf(UBIDOTS_DEBUG "Error populating hostname buffer\n");
668  memset(conn.host_header, 0, UBIDOTS_HOST_HEADER_LEN);
669  return;
670  }
671 
672  remaining -= host_len;
673 
674  if(UBIDOTS_REMOTE_PORT != 80) {
675  host_len = snprintf(&conn.host_header[host_len], remaining,
676  ":%u", UBIDOTS_REMOTE_PORT);
677 
678  if(host_len < 0 || host_len >= remaining) {
679  printf(UBIDOTS_DEBUG "Error populating hostname buffer\n");
680  memset(conn.host_header, 0, UBIDOTS_HOST_HEADER_LEN);
681  return;
682  }
683  remaining -= host_len;
684  }
685 
686  PRINTF(UBIDOTS_DEBUG "'Host: %s' (remaining %u)\n", conn.host_header,
687  remaining);
688 }

726 /*---------------------------------------------------------------------------*/
727 static void
728 state_machine(void)
729 {
730 #if UBIDOTS_DNS_SUPPORT
731  uip_ipaddr_t *tmp_addr_ptr = NULL;
732  resolv_status_t resolv_status;
733 #endif
734 
735  switch(conn.state) {
736  case STATE_ERROR_NO_NET:
737  PRINTF(UBIDOTS_DEBUG "STATE_ERROR_NO_NET\n");
738  if(have_network() == 0) {
739  break;
740  }
741 
742  conn.state = STATE_STARTING;
743  /* Continue */
744  case STATE_STARTING:
745  PRINTF(UBIDOTS_DEBUG "STATE_STARTING\n");
746  conn.connection_attempt = 0;
747 
748  /*
749  * First, check whether UBIDOTS_REMOTE_HOST is a string representation of
750  * an IP address. If not, we need to resolve it over DNS
751  */
752  PRINTF(UBIDOTS_DEBUG "Checking " UBIDOTS_REMOTE_HOST "\n");
753 
754  if(ip_addr_conv(UBIDOTS_REMOTE_HOST, &conn.remote_ip) == 0) {
755 #if UBIDOTS_DNS_SUPPORT
756  /*
757  * The address was not parsed, so we assume it is a hostname. Enter
758  * STATE_RESOLVING to attempt resolution over DNS
759  */
760  conn.state = STATE_RESOLVING;
761 
762  /* Populate the buffer which will be used for the HTTP Host: header */
763  encode_http_host(0);
764 #else
765  /*
766  * Could not convert UBIDOTS_REMOTE_HOST to an IP address and we do not
767  * have UDP support, therefore we cannot resolve over DNS.
768  *
769  * This is fatal and we cannot recover. The process will exit.
770  */
771  conn.state = STATE_ERROR_INVALID_REMOTE;
772 #endif
773  } else {
774  conn.state = STATE_TCP_CONNECT;
775 
776  /* Populate the buffer which will be used for the HTTP Host: header */
777  encode_http_host(1);
778  }
779 
780  /*
781  * If we reach here, we can be in one of the following states:
782  * - STATE_RESOLVING
783  * - STATE_TCP_CONNECT
784  * - STATE_ERROR_INVALID_REMOTE
785  * In all cases, continue immediately
786  */
787  process_poll(&ubidots_client_process);
788  return;
789  case STATE_DISCONNECTED:
790  PRINTF(UBIDOTS_DEBUG "STATE_DISCONNECTED\n");
791  process_post(conn.consumer, ubidots_event_disconnected, NULL);
792  etimer_set(&periodic, UBIDOTS_RECONNECT_INTERVAL);
793  conn.state = STATE_TCP_CONNECT;
794  return;
795 #if UBIDOTS_DNS_SUPPORT
796  /* This state is only relevant if we have DNS support */
797  case STATE_RESOLVING:
798  resolv_status = resolv_lookup(UBIDOTS_REMOTE_HOST, &tmp_addr_ptr);
799 
800  PRINTF(UBIDOTS_DEBUG "STATE_RESOLVING (%u)\n", resolv_status);
801 
802  if(resolv_status == RESOLV_STATUS_CACHED) {
803  /*
804  * An entry for the hostname was found in the resolver's cache, and
805  * it's fresh and usable. Copy over and continue
806  */
807  PRINTF(UBIDOTS_DEBUG "Entry for " UBIDOTS_REMOTE_HOST " cached and usable.\n");
808  memcpy(&conn.remote_ip, tmp_addr_ptr, sizeof(uip_ipaddr_t));
809  conn.state = STATE_TCP_CONNECT;
810  process_poll(&ubidots_client_process);
811  return;
812  } else if(resolv_status == RESOLV_STATUS_EXPIRED ||
813  resolv_status == RESOLV_STATUS_UNCACHED) {
814  PRINTF(UBIDOTS_DEBUG "Resolving " UBIDOTS_REMOTE_HOST "\n");
815  resolv_query(UBIDOTS_REMOTE_HOST);
816  }
817 
818  /* RESOLV_STATUS_RESOLVING, _NOT_FOUND, _ERROR: wait */
819  break;
820 #endif
821  case STATE_TCP_CONNECT:
822  conn.connection_attempt++;
823 
824  PRINTF(UBIDOTS_DEBUG "STATE_TCP_CONNECT (%u)\n", conn.connection_attempt);
825 
826  if(conn.connection_attempt <= UBIDOTS_CONNECT_ATTEMPTS ||
827  UBIDOTS_CONNECT_ATTEMPTS == RETRY_FOREVER) {
828  if(tcp_conn_connect() == 1) {
829  /*
830  * Here we simply return and we wait for the TCP socket to generate
831  * an event for us. event_callback() will put us in the correct state
832  * and poll us.
833  */
834  return;
835  } else {
836  /*
837  * tcp_conn_connect() returned an error, this means that
838  * tcp_socket_connect() returned an error. This is fatal.
839  */
840  conn.state = STATE_ERROR_TCP_SOCKET;
841  }
842  } else {
843  /* Exceeded max connection attempts */
844  conn.state = STATE_ERROR_MAX_ATTEMPTS;
845  }
846  break;
847  case STATE_TCP_CONNECTED:
848  PRINTF(UBIDOTS_DEBUG "STATE_TCP_CONNECTED\n");
849 
850  /*
851  * Set maximum size for this socket's outgoing TCP segments
852  * Low: Avoid or decrease 6lowpan fragmentation
853  * High: Reduce TCP data/ACK back and forth
854  */
855  conn.sock.output_data_max_seg = UBIDOTS_MAX_TCP_SEG;
856  conn.connection_attempt = 0;
857  conn.state = STATE_ESTABLISHED;
858 
859  process_post(conn.consumer, ubidots_event_established, NULL);
860  break;
861  case STATE_ESTABLISHED:
862  /* Idle state. We wait for the consumer to do something */
863  break;
864  case STATE_POSTING:
865  PRINTF(UBIDOTS_DEBUG "STATE_POSTING (%u)\n", conn.sock.output_data_len);
866  if(conn.sock.output_data_len == 0) {
867  process_post(conn.consumer, ubidots_event_post_sent, NULL);
868  conn.state = STATE_ESTABLISHED;
869  }
870  break;
871  case STATE_ERROR_MAX_ATTEMPTS:
872  case STATE_ERROR_TCP_SOCKET:
873  case STATE_ERROR_INVALID_REMOTE:
874  case STATE_ERROR_UNKNOWN:
875  default:
876  printf("State 0x%02x. This is fatal and the client will exit.\n",
877  conn.state);
878  /*
879  * These are all hard error states that we cannot recover from.
880  * Default should never happen
881  */
882  ubidots_conn_teardown();
883  ubidots_conn_reset();
884  process_exit(&ubidots_client_process);
885  return;
886  }
887 
888  etimer_set(&periodic, STATE_MACHINE_PERIODIC);
889 }

892 ubidots_enqueue_value(const char *variable, const char *value)
893 {
894  /* Add a JSON comma separator, except before the opening { */
895  if(conn.out_buf[conn.out_buf_pos - 1] != '[') {
896  if(enqueue_chunk(",") == UBIDOTS_ERROR) {
897  return UBIDOTS_ERROR;
898  }
899  }
900 
901  if(enqueue_chunk("{") == UBIDOTS_ERROR) {
902  return UBIDOTS_ERROR;
903  }
904 
905  if(variable) {
906  if(enqueue_chunk("\"variable\":\"%s\",", variable) == UBIDOTS_ERROR) {
907  return UBIDOTS_ERROR;
908  }
909  }
910 
911  if(enqueue_chunk("\"value\":%s}", value) == UBIDOTS_ERROR) {
912  return UBIDOTS_ERROR;
913  }
914 
915  PRINTF(UBIDOTS_DEBUG "Enqueue value: Buffer at %u\n", conn.out_buf_pos);
916 
917  return UBIDOTS_SUCCESS;
918 }

944 /*---------------------------------------------------------------------------*/
945 int
946 ubidots_post()
947 {
948  uint16_t content_length;
949  uint8_t content_len_len;
950  uint16_t move_from;
951  uint16_t move_to;
952  uint16_t move_count;
953 
954  if(enqueue_chunk("]") == UBIDOTS_ERROR) {
955  return UBIDOTS_ERROR;
956  }
957 
958  /* Calculate and fill in Content-Length: */
959  content_length = conn.out_buf_pos - conn.payload_start_pos;
960  content_len_len = snprintf((char *)(&conn.out_buf[conn.content_length_pos]),
961  HEADER_CONTENT_LEN_STR_LEN, "%u", content_length);
962 
963  PRINTF(UBIDOTS_DEBUG "POST: Buffer at %u, content-length %u (%u), at %u\n",
964  conn.out_buf_pos, content_length, content_len_len,
965  conn.content_length_pos);
966 
967  /*
968  * Pedantry: Slide the remainder of the POST left, to remove the spaces
969  * that may be trailing Content-Length:. snprintf will have also left a NULL
970  * character in there, which we also want to get rid of in the process
971  *
972  * We occupied content_len_len of HEADER_CONTENT_LEN_STR_LEN bytes
973  * Move:
974  * from conn.content_length_pos + HEADER_CONTENT_LEN_STR_LEN
975  * to conn.content_length_pos + content_len_len:
976  * count: conn.out_buf_pos - conn.content_length_pos - (move_from - move_to)
977  */
978  if(content_len_len > 0 && content_len_len < HEADER_CONTENT_LEN_STR_LEN) {
979  move_to = conn.content_length_pos + content_len_len;
980  move_from = conn.content_length_pos + HEADER_CONTENT_LEN_STR_LEN;
981  move_count = conn.out_buf_pos - conn.content_length_pos -
982  (move_from - move_to);
983  memmove((char *)(&conn.out_buf[move_to]),
984  (char *)(&conn.out_buf[move_from]),
985  move_count);
986  conn.out_buf_pos -= HEADER_CONTENT_LEN_STR_LEN - content_len_len;
987  }
988 
989  tcp_socket_send(&conn.sock, conn.out_buf, conn.out_buf_pos);
990 
991  conn.state = STATE_POSTING;
992 
993  PRINTF(UBIDOTS_DEBUG "POST: Buffer at %u\n", conn.out_buf_pos);
994 
995  return UBIDOTS_SUCCESS;
996 }
