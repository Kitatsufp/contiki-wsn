97 /**
98  * XORs \p n bytes byte-by-byte starting at \p y to the memory area
99  * starting at \p x. */
100 static inline void
101 memxor(unsigned char *x, const unsigned char *y, size_t n) {
102  while(n--) {
103  *x ^= *y;
104  x++; y++;
105  }
106 }

108 /**
109  * Compares \p len bytes from @p a with @p b in constant time. This
110  * functions always traverses the entire length to prevent timing
111  * attacks.
112  *
113  * \param a Byte sequence to compare
114  * \param b Byte sequence to compare
115  * \param len Number of bytes to compare.
116  * \return \c 1 if \p a and \p b are equal, \c 0 otherwise.
117  */
118 static inline int
119 equals(unsigned char *a, unsigned char *b, size_t len) {
120  int result = 1;
121  while (len--) {
122  result &= (*a++ == *b++);
123  }
124  return result;
125 }
