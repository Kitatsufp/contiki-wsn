50  /*
51  To get rf230bb radio in sniff mode we need to have radio in RX_ON.

54  a chance other radios works the same way.
55  */
56 
57  rf230_set_promiscuous_mode(1);
58 
59  printf("Sniffer started\n");

61  while(1) {
