222 int
223 dtls_write(struct dtls_context_t *ctx,
224  session_t *dst, uint8 *buf, size_t len) {
225 
226  dtls_peer_t *peer = dtls_get_peer(ctx, dst);
227 
228  /* Check if peer connection already exists */
229  if (!peer) { /* no ==> create one */
230  int res;
231 
232  /* dtls_connect() returns a value greater than zero if a new
233  * connection attempt is made, 0 for session reuse. */
234  res = dtls_connect(ctx, dst);
235 
236  return (res >= 0) ? 0 : res;
237  } else { /* a session exists, check if it is in state connected */
238 
239  if (peer->state != DTLS_STATE_CONNECTED) {
240  return 0;
241  } else {
242  return dtls_send(ctx, peer, DTLS_CT_APPLICATION_DATA, buf, len);
243  }
244  }

247 static int
248 dtls_get_cookie(uint8 *msg, size_t msglen, uint8 **cookie) {
249  /* To access the cookie, we have to determine the session id's
250  * length and skip the whole thing. */
251  if (msglen < DTLS_HS_LENGTH + DTLS_CH_LENGTH + sizeof(uint8))
252  return dtls_alert_fatal_create(DTLS_ALERT_HANDSHAKE_FAILURE);
253 
254  if (dtls_uint16_to_int(msg + DTLS_HS_LENGTH) != DTLS_VERSION)
255  return dtls_alert_fatal_create(DTLS_ALERT_PROTOCOL_VERSION);
256 
257  msglen -= DTLS_HS_LENGTH + DTLS_CH_LENGTH;
258  msg += DTLS_HS_LENGTH + DTLS_CH_LENGTH;
259 
260  SKIP_VAR_FIELD(msg, msglen, uint8); /* skip session id */
261 
262  if (msglen < (*msg & 0xff) + sizeof(uint8))
263  return dtls_alert_fatal_create(DTLS_ALERT_HANDSHAKE_FAILURE);
264 
265  *cookie = msg + sizeof(uint8);
266  return dtls_uint8_to_int(msg);
267 
268  error:
269  return dtls_alert_fatal_create(DTLS_ALERT_HANDSHAKE_FAILURE);
270 }

272 static int
273 dtls_create_cookie(dtls_context_t *ctx,
274  session_t *session,
275  uint8 *msg, size_t msglen,
276  uint8 *cookie, int *clen) {
277  unsigned char buf[DTLS_HMAC_MAX];
278  size_t len, e;
279 
280  /* create cookie with HMAC-SHA256 over:
281  * - SECRET
282  * - session parameters (only IP address?)
283  * - client version
284  * - random gmt and bytes
285  * - session id
286  * - cipher_suites
287  * - compression method
288  */
289 
290  /* We use our own buffer as hmac_context instead of a dynamic buffer
291  * created by dtls_hmac_new() to separate storage space for cookie
292  * creation from storage that is used in real sessions. Note that
293  * the buffer size must fit with the default hash algorithm (see
294  * implementation of dtls_hmac_context_new()). */
295 
296  dtls_hmac_context_t hmac_context;
297  dtls_hmac_init(&hmac_context, ctx->cookie_secret, DTLS_COOKIE_SECRET_LENGTH);
298 
299  dtls_hmac_update(&hmac_context,
300  (unsigned char *)&session->addr, session->size);
301 
302  /* feed in the beginning of the Client Hello up to and including the
303  session id */
304  e = sizeof(dtls_client_hello_t);
305  e += (*(msg + DTLS_HS_LENGTH + e) & 0xff) + sizeof(uint8);
306  if (e + DTLS_HS_LENGTH > msglen)
307  return dtls_alert_fatal_create(DTLS_ALERT_HANDSHAKE_FAILURE);
308 
309  dtls_hmac_update(&hmac_context, msg + DTLS_HS_LENGTH, e);
310 
311  /* skip cookie bytes and length byte */
312  e += *(uint8 *)(msg + DTLS_HS_LENGTH + e) & 0xff;
313  e += sizeof(uint8);
314  if (e + DTLS_HS_LENGTH > msglen)
315  return dtls_alert_fatal_create(DTLS_ALERT_HANDSHAKE_FAILURE);
316 
317  dtls_hmac_update(&hmac_context,
318  msg + DTLS_HS_LENGTH + e,
319  dtls_get_fragment_length(DTLS_HANDSHAKE_HEADER(msg)) - e);
320 
321  len = dtls_hmac_finalize(&hmac_context, buf);
322 
323  if (len < *clen) {
324  memset(cookie + len, 0, *clen - len);
325  *clen = len;
326  }
327 
328  memcpy(cookie, buf, *clen);
329  return 0;
330 }

369 /**
370  * Initializes \p buf as record header. The caller must ensure that \p
371  * buf is capable of holding at least \c sizeof(dtls_record_header_t)
372  * bytes. Increments sequence number counter of \p security.
373  * \return pointer to the next byte after the written header.
374  * The length will be set to 0 and has to be changed before sending.
375  */
376 static inline uint8 *
377 dtls_set_record_header(uint8 type, dtls_security_parameters_t *security,
378  uint8 *buf) {
379 
380  dtls_int_to_uint8(buf, type);
381  buf += sizeof(uint8);
382 
383  dtls_int_to_uint16(buf, DTLS_VERSION);
384  buf += sizeof(uint16);
385 
386  if (security) {
387  dtls_int_to_uint16(buf, security->epoch);
388  buf += sizeof(uint16);
389 
390  dtls_int_to_uint48(buf, security->rseq);
391  buf += sizeof(uint48);
392 
393  /* increment record sequence counter by 1 */
394  security->rseq++;
395  } else {
396  memset(buf, 0, sizeof(uint16) + sizeof(uint48));
397  buf += sizeof(uint16) + sizeof(uint48);
398  }
399 
400  memset(buf, 0, sizeof(uint16));
401  return buf + sizeof(uint16);
402 }

404 /**
405  * Initializes \p buf as handshake header. The caller must ensure that \p
406  * buf is capable of holding at least \c sizeof(dtls_handshake_header_t)
407  * bytes. Increments message sequence number counter of \p peer.
408  * \return pointer to the next byte after \p buf
409  */
410 static inline uint8 *
411 dtls_set_handshake_header(uint8 type, dtls_peer_t *peer,
412  int length,
413  int frag_offset, int frag_length,
414  uint8 *buf) {
415 
416  dtls_int_to_uint8(buf, type);
417  buf += sizeof(uint8);
418 
419  dtls_int_to_uint24(buf, length);
420  buf += sizeof(uint24);
421 
422  if (peer && peer->handshake_params) {
423  /* and copy the result to buf */
424  dtls_int_to_uint16(buf, peer->handshake_params->hs_state.mseq_s);
425 
426  /* increment handshake message sequence counter by 1 */
427  peer->handshake_params->hs_state.mseq_s++;
428  } else {
429  memset(buf, 0, sizeof(uint16));
430  }
431  buf += sizeof(uint16);
432 
433  dtls_int_to_uint24(buf, frag_offset);
434  buf += sizeof(uint24);
435 
436  dtls_int_to_uint24(buf, frag_length);
437  buf += sizeof(uint24);
438 
439  return buf;
440 }

519 /**
520  * This method detects if we already have a established DTLS session with
521  * peer and the peer is attempting to perform a fresh handshake by sending
522  * messages with epoch = 0. This is to handle situations mentioned in
523  * RFC 6347 - section 4.2.8.
524  *
525  * @param msg The packet received from Client
526  * @param msglen Packet length
527  * @param peer peer who is the sender for this packet
528  * @return @c 1 if this is a rehandshake attempt by
529  * client
530  */
531 static int
532 hs_attempt_with_existing_peer(uint8_t *msg, size_t msglen,
533  dtls_peer_t *peer)
534 {
535  if ((peer) && (peer->state == DTLS_STATE_CONNECTED)) {
536  if (msg[0] == DTLS_CT_HANDSHAKE) {
537  uint16_t msg_epoch = dtls_uint16_to_int(DTLS_RECORD_HEADER(msg)->epoch);
538  if (msg_epoch == 0) {
539  dtls_handshake_header_t * hs_header = DTLS_HANDSHAKE_HEADER(msg + DTLS_RH_LENGTH);
540  if (hs_header->msg_type == DTLS_HT_CLIENT_HELLO ||
541  hs_header->msg_type == DTLS_HT_HELLO_REQUEST) {
542  return 1;
543  }
544  }
545  }
546  }
547  return 0;
548 }

613  * Calculate the pre master secret and after that calculate the master-secret.
614  */
615 static int
616 calculate_key_block(dtls_context_t *ctx,
617  dtls_handshake_parameters_t *handshake,
618  dtls_peer_t *peer,
619  session_t *session,
620  dtls_peer_type role) {
621  unsigned char *pre_master_secret;
622  int pre_master_len = 0;
623  dtls_security_parameters_t *security = dtls_security_params_next(peer);
624  uint8 master_secret[DTLS_MASTER_SECRET_LENGTH];
625 
626  if (!security) {
627  return dtls_alert_fatal_create(DTLS_ALERT_INTERNAL_ERROR);
628  }
629 
630  pre_master_secret = security->key_block;
631 
632  switch (handshake->cipher) {
633 #ifdef DTLS_PSK
634  case TLS_PSK_WITH_AES_128_CCM_8: {
635  unsigned char psk[DTLS_PSK_MAX_KEY_LEN];
636  int len;
637 
638  len = CALL(ctx, get_psk_info, session, DTLS_PSK_KEY,
639  handshake->keyx.psk.identity,
640  handshake->keyx.psk.id_length,
641  psk, DTLS_PSK_MAX_KEY_LEN);
642  if (len < 0) {
643  dtls_crit("no psk key for session available\n");
644  return len;
645  }
646  /* Temporarily use the key_block storage space for the pre master secret. */
647  pre_master_len = dtls_psk_pre_master_secret(psk, len,
648  pre_master_secret,
649  MAX_KEYBLOCK_LENGTH);
650 
651  dtls_debug_hexdump("psk", psk, len);
652 
653  memset(psk, 0, DTLS_PSK_MAX_KEY_LEN);
654  if (pre_master_len < 0) {
655  dtls_crit("the psk was too long, for the pre master secret\n");
656  return dtls_alert_fatal_create(DTLS_ALERT_INTERNAL_ERROR);
657  }
658 
659  break;
660  }
661 #endif /* DTLS_PSK */
662 #ifdef DTLS_ECC
663  case TLS_ECDHE_ECDSA_WITH_AES_128_CCM_8: {
664  pre_master_len = dtls_ecdh_pre_master_secret(handshake->keyx.ecdsa.own_eph_priv,
665  handshake->keyx.ecdsa.other_eph_pub_x,
666  handshake->keyx.ecdsa.other_eph_pub_y,
667  sizeof(handshake->keyx.ecdsa.own_eph_priv),
668  pre_master_secret,
669  MAX_KEYBLOCK_LENGTH);
670  if (pre_master_len < 0) {
671  dtls_crit("the curve was too long, for the pre master secret\n");
672  return dtls_alert_fatal_create(DTLS_ALERT_INTERNAL_ERROR);
673  }
674  break;
675  }
676 #endif /* DTLS_ECC */
677  default:
678  dtls_crit("calculate_key_block: unknown cipher\n");
679  return dtls_alert_fatal_create(DTLS_ALERT_INTERNAL_ERROR);
680  }

682  dtls_debug_dump("client_random", handshake->tmp.random.client, DTLS_RANDOM_LENGTH);
683  dtls_debug_dump("server_random", handshake->tmp.random.server, DTLS_RANDOM_LENGTH);
684  dtls_debug_dump("pre_master_secret", pre_master_secret, pre_master_len);
685

687  PRF_LABEL(master), PRF_LABEL_SIZE(master),
688  handshake->tmp.random.client, DTLS_RANDOM_LENGTH,
689  handshake->tmp.random.server, DTLS_RANDOM_LENGTH,
690  master_secret,
691  DTLS_MASTER_SECRET_LENGTH);
692 
693  dtls_debug_dump("master_secret", master_secret, DTLS_MASTER_SECRET_LENGTH);
694 
695  /* create key_block from master_secret
696  * key_block = PRF(master_secret,
697  "key expansion" + tmp.random.server + tmp.random.client) */
698 
699  dtls_prf(master_secret,
700  DTLS_MASTER_SECRET_LENGTH,
701  PRF_LABEL(key), PRF_LABEL_SIZE(key),
702  handshake->tmp.random.server, DTLS_RANDOM_LENGTH,
703  handshake->tmp.random.client, DTLS_RANDOM_LENGTH,
704  security->key_block,
705  dtls_kb_size(security, role));
706 
707  memcpy(handshake->tmp.master_secret, master_secret, DTLS_MASTER_SECRET_LENGTH);
708  dtls_debug_keyblock(security);
709 
710  security->cipher = handshake->cipher;
711  security->compression = handshake->compression;
712  security->rseq = 0;
713 
714  return 0;
715 }

790 /*
791  * Check for some TLS Extensions used by the ECDHE_ECDSA cipher.
792  */
793 static int
794 dtls_check_tls_extension(dtls_peer_t *peer,
795  uint8 *data, size_t data_length, int client_hello)
796 {
797  uint16_t i, j;
798  int ext_elliptic_curve = 0;
799  int ext_client_cert_type = 0;
800  int ext_server_cert_type = 0;
801  int ext_ec_point_formats = 0;
802  dtls_handshake_parameters_t *handshake = peer->handshake_params;
803 
804  if (data_length < sizeof(uint16)) {
805  /* no tls extensions specified */
806  if (is_tls_ecdhe_ecdsa_with_aes_128_ccm_8(handshake->cipher)) {
807  goto error;
808  }
809  return 0;
810  }
811 
812  /* get the length of the tls extension list */
813  j = dtls_uint16_to_int(data);
814  data += sizeof(uint16);
815  data_length -= sizeof(uint16);
816 
817  if (data_length < j)
818  goto error;
819 
820  /* check for TLS extensions needed for this cipher */
821  while (data_length) {
822  if (data_length < sizeof(uint16) * 2)
823  goto error;
824 
825  /* get the tls extension type */
826  i = dtls_uint16_to_int(data);
827  data += sizeof(uint16);
828  data_length -= sizeof(uint16);
829 
830  /* get the length of the tls extension */
831  j = dtls_uint16_to_int(data);
832  data += sizeof(uint16);
833  data_length -= sizeof(uint16);
834 
835  if (data_length < j)
836  goto error;
837 
838  switch (i) {
839  case TLS_EXT_ELLIPTIC_CURVES:
840  ext_elliptic_curve = 1;
841  if (verify_ext_eliptic_curves(data, j))
842  goto error;
843  break;
844  case TLS_EXT_CLIENT_CERTIFICATE_TYPE:
845  ext_client_cert_type = 1;
846  if (client_hello) {
847  if (verify_ext_cert_type(data, j))
848  goto error;
849  } else {
850  if (dtls_uint8_to_int(data) != TLS_CERT_TYPE_RAW_PUBLIC_KEY)
851  goto error;
852  }
853  break;
854  case TLS_EXT_SERVER_CERTIFICATE_TYPE:
855  ext_server_cert_type = 1;
856  if (client_hello) {
857  if (verify_ext_cert_type(data, j))
858  goto error;
859  } else {
860  if (dtls_uint8_to_int(data) != TLS_CERT_TYPE_RAW_PUBLIC_KEY)
861  goto error;
862  }
863  break;
864  case TLS_EXT_EC_POINT_FORMATS:
865  ext_ec_point_formats = 1;
866  if (verify_ext_ec_point_formats(data, j))
867  goto error;
868  break;
869  case TLS_EXT_ENCRYPT_THEN_MAC:
870  /* As only AEAD cipher suites are currently available, this
871  * extension can be skipped.
872  */
873  dtls_info("skipped encrypt-then-mac extension\n");
874  break;
875  default:
876  dtls_warn("unsupported tls extension: %i\n", i);
877  break;
878  }
879  data += j;
880  data_length -= j;
881  }
882  if (is_tls_ecdhe_ecdsa_with_aes_128_ccm_8(handshake->cipher) && client_hello) {
883  if (!ext_elliptic_curve || !ext_client_cert_type || !ext_server_cert_type
884  || !ext_ec_point_formats) {
885  dtls_warn("not all required tls extensions found in client hello\n");
886  goto error;
887  }
888  } else if (is_tls_ecdhe_ecdsa_with_aes_128_ccm_8(handshake->cipher) && !client_hello) {
889  if (!ext_client_cert_type || !ext_server_cert_type) {
890  dtls_warn("not all required tls extensions found in server hello\n");
891  goto error;
892  }
893  }
894  return 0;
895 
896 error:
897  if (client_hello && peer->state == DTLS_STATE_CONNECTED) {
898  return dtls_alert_create(DTLS_ALERT_LEVEL_WARNING, DTLS_ALERT_NO_RENEGOTIATION);
899  } else {
900  return dtls_alert_fatal_create(DTLS_ALERT_HANDSHAKE_FAILURE);
901  }
902 }

904 /**
905  * Parses the ClientHello from the client and updates the internal handshake
906  * parameters with the new data for the given \p peer. When the ClientHello
907  * handshake message in \p data does not contain a cipher suite or
908  * compression method, it is copied from the the current security parameters.
909  *
910  * \param ctx The current DTLS context.
911  * \param peer The remote peer whose security parameters are about to change.
912  * \param data The handshake message with a ClientHello.
913  * \param data_length The actual size of \p data.
914  * \return \c -Something if an error occurred, \c 0 on success.
915  */
916 static int
917 dtls_update_parameters(dtls_context_t *ctx,
918  dtls_peer_t *peer,
919  uint8 *data, size_t data_length) {
920  int i, j;
921  int ok;
922  dtls_handshake_parameters_t *config = peer->handshake_params;
923  dtls_security_parameters_t *security = dtls_security_params(peer);
924 
925  assert(config);
926  assert(data_length > DTLS_HS_LENGTH + DTLS_CH_LENGTH);
927 
928  /* skip the handshake header and client version information */
929  data += DTLS_HS_LENGTH + sizeof(uint16);
930  data_length -= DTLS_HS_LENGTH + sizeof(uint16);
931 
932  /* store client random in config */
933  memcpy(config->tmp.random.client, data, DTLS_RANDOM_LENGTH);
934  data += DTLS_RANDOM_LENGTH;
935  data_length -= DTLS_RANDOM_LENGTH;
936 
937  /* Caution: SKIP_VAR_FIELD may jump to error: */
938  SKIP_VAR_FIELD(data, data_length, uint8); /* skip session id */
939  SKIP_VAR_FIELD(data, data_length, uint8); /* skip cookie */
940 
941  i = dtls_uint16_to_int(data);
942  if (data_length < i + sizeof(uint16)) {
943  /* Looks like we do not have a cipher nor compression. This is ok
944  * for renegotiation, but not for the initial handshake. */
945 
946  if (!security || security->cipher == TLS_NULL_WITH_NULL_NULL)
947  goto error;
948 
949  config->cipher = security->cipher;
950  config->compression = security->compression;
951 
952  return 0;
953  }
954 
955  data += sizeof(uint16);
956  data_length -= sizeof(uint16) + i;
957 
958  ok = 0;
959  while (i && !ok) {
960  config->cipher = dtls_uint16_to_int(data);
961  ok = known_cipher(ctx, config->cipher, 0);
962  i -= sizeof(uint16);
963  data += sizeof(uint16);
964  }
965 
966  /* skip remaining ciphers */
967  data += i;
968 
969  if (!ok) {
970  /* reset config cipher to a well-defined value */
971  config->cipher = TLS_NULL_WITH_NULL_NULL;
972  dtls_warn("No matching cipher found\n");
973  goto error;
974  }
975 
976  if (data_length < sizeof(uint8)) {
977  /* no compression specified, take the current compression method */
978  if (security)
979  config->compression = security->compression;
980  else
981  config->compression = TLS_COMPRESSION_NULL;
982  return 0;
983  }
984 
985  i = dtls_uint8_to_int(data);
986  if (data_length < i + sizeof(uint8))
987  goto error;
988 
989  data += sizeof(uint8);
990  data_length -= sizeof(uint8) + i;
991 
992  ok = 0;
993  while (i && !ok) {
994  for (j = 0; j < sizeof(compression_methods) / sizeof(uint8); ++j)
995  if (dtls_uint8_to_int(data) == compression_methods[j]) {
996  config->compression = compression_methods[j];
997  ok = 1;
998  }
999  i -= sizeof(uint8);
1000  data += sizeof(uint8);
1001  }
1002 
1003  if (!ok) {
1004  /* reset config cipher to a well-defined value */
1005  goto error;
1006  }
1007 
1008  return dtls_check_tls_extension(peer, data, data_length, 1);
1009 error:
1010  if (peer->state == DTLS_STATE_CONNECTED) {
1011  return dtls_alert_create(DTLS_ALERT_LEVEL_WARNING, DTLS_ALERT_NO_RENEGOTIATION);
1012  } else {
1013  return dtls_alert_fatal_create(DTLS_ALERT_HANDSHAKE_FAILURE);
1014  }
1015 }

1121 check_finished(dtls_context_t *ctx, dtls_peer_t *peer,
1122  uint8 *data, size_t data_length) {

1124  const unsigned char *label;
1125  unsigned char buf[DTLS_HMAC_MAX];
1126

1128  return dtls_alert_fatal_create(DTLS_ALERT_HANDSHAKE_FAILURE);

1149  label = PRF_LABEL(server);
1150  label_size = PRF_LABEL_SIZE(server);
1151  } else { /* server */

1153  label_size = PRF_LABEL_SIZE(client);

1157  DTLS_MASTER_SECRET_LENGTH,

1159  PRF_LABEL(finished), PRF_LABEL_SIZE(finished),

1161  b.verify_data, sizeof(b.verify_data));

1232  * seq_num(2+6) + type(1) + version(2) + length(2)

1242  } else {

1251  3.2.1 in [RFC5116]).

1255  opaque nonce_explicit[8];
1256  } CCMNonce;
1257 
1258  [...]

1261  48-bit seq_num.
1262 
1263  When the nonce_explicit is equal to the sequence number, the CCMNonce
1264  will have the structure of the CCMNonceExample given below.
1265 
1266  struct {
1267  uint32 client_write_IV; // low order 32-bits
1268  uint64 seq_num; // TLS sequence number
1269  } CCMClientNonce.
1270 
1271 
1272  struct {
1273  uint32 server_write_IV; // low order 32-bits
1274  uint64 seq_num; // TLS sequence number
1275  } CCMServerNonce.
1276 
1277 
1278  struct {
1279  case client:
1280  CCMClientNonce;
1281  case server:
1282  CCMServerNonce:
1283  } CCMNonceExample;
1284  */

1287  p += 8;
1288  res = 8;
1289 
1290  for (i = 0; i < data_array_len; i++) {
1291  /* check the minimum that we need for packets that are not encrypted */
1292  if (*rlen < res + DTLS_RH_LENGTH + data_len_array[i]) {
1293  dtls_debug("dtls_prepare_record: send buffer too small\n");
1294  return dtls_alert_fatal_create(DTLS_ALERT_INTERNAL_ERROR);
1295  }
1296 
1297  memcpy(p, data_array[i], data_len_array[i]);
1298  p += data_len_array[i];
1299  res += data_len_array[i];
1300  }
1301 
1302  memset(nonce, 0, DTLS_CCM_BLOCKSIZE);
1303  memcpy(nonce, dtls_kb_local_iv(security, peer->role),
1304  dtls_kb_iv_size(security, peer->role));
1305  memcpy(nonce + dtls_kb_iv_size(security, peer->role), start, 8); /* epoch + seq_num */
1306 
1307  dtls_debug_dump("nonce:", nonce, DTLS_CCM_BLOCKSIZE);
1308  dtls_debug_dump("key:", dtls_kb_local_write_key(security, peer->role),
1309  dtls_kb_key_size(security, peer->role));
1310 
1311  /* re-use N to create additional data according to RFC 5246, Section 6.2.3.3:
1312  *
1313  * additional_data = seq_num + TLSCompressed.type +
1314  * TLSCompressed.version + TLSCompressed.length;
1315  */
1316  memcpy(A_DATA, &DTLS_RECORD_HEADER(sendbuf)->epoch, 8); /* epoch and seq_num */
1317  memcpy(A_DATA + 8, &DTLS_RECORD_HEADER(sendbuf)->content_type, 3); /* type and version */
1318  dtls_int_to_uint16(A_DATA + 11, res - 8); /* length */

1321  dtls_kb_local_write_key(security, peer->role),
1322  dtls_kb_key_size(security, peer->role),
1323  A_DATA, A_DATA_LEN);
1324 
1325  if (res < 0)
1326  return res;
1327 
1328  res += 8; /* increment res by size of nonce_explicit */
1329  dtls_debug_dump("message:", start, res);
1330  }
1331 
1332  /* fix length of fragment in sendbuf */
1333  dtls_int_to_uint16(sendbuf + 11, res);
1334 
1335  *rlen = DTLS_RH_LENGTH + res;
1336  return 0;
1337 }
