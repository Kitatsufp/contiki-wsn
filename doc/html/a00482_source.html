75 /**
76  * Context for HMAC generation. This object is initialized with
77  * dtls_hmac_init() and must be passed to dtls_hmac_update() and
78  * dtls_hmac_finalize(). Once, finalized, the component \c H is
79  * invalid and must be initialized again with dtls_hmac_init() before
80  * the structure can be used again.
81  */

84  dtls_hash_ctx data; /**< context for hash function */
