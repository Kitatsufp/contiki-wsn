51 netq_insert_node(netq_t **queue, netq_t *node) {
52  netq_t *p;
53 
54  assert(queue);
55  assert(node);
56 
57  p = *queue;
58  while(p && p->t <= node->t)
59  p = p->next;
60 
61  if (p)
62  LL_PREPEND_ELEM(*queue, p, node);
63  else
64  LL_APPEND(*queue, node);
65 
66  return 1;
67 }
