63 /**
64  * This structure contains callback functions used by tinydtls to
65  * communicate with the application. At least the write function must
66  * be provided. It is called by the DTLS state machine to send packets
67  * over the network. The read function is invoked to deliver decrypted
68  * and verfified application data. The third callback is an event
69  * handler function that is called when alert messages are encountered
70  * or events generated by the library have occured.
71  */

75  * network. The callback function must use the network interface
76  * denoted by session->ifindex to send the data.
77  *
78  * @param ctx The current DTLS context.
79  * @param session The session object, including the address of the

84  * that were sent, or a value less than zero to indicate an

420  *
421  * This library provides a very simple datagram server with DTLS
422  * support. It is designed to support session multiplexing in
423  * single-threaded applications and thus targets specifically on
424  * embedded systems.
425  *
426  * @section license License
427  *
428  * This software is under the <a
429  * href="http://www.opensource.org/licenses/mit-license.php">MIT License</a>.
430  *
431  * @subsection uthash UTHash
432  *
433  * This library uses <a href="http://uthash.sourceforge.net/">uthash</a> to manage
434  * its peers (not used for Contiki). @b uthash uses the <b>BSD revised license</b>, see
435  * <a href="http://uthash.sourceforge.net/license.html">http://uthash.sourceforge.net/license.html</a>.
436  *
437  * @subsection sha256 Aaron D. Gifford's SHA256 Implementation
438  *
439  * tinyDTLS provides HMAC-SHA256 with BSD-licensed code from Aaron D. Gifford,
440  * see <a href="http://www.aarongifford.com/">www.aarongifford.com</a>.
441  *
442  * @subsection aes Rijndael Implementation From OpenBSD
443  *
444  * The AES implementation is taken from rijndael.{c,h} contained in the crypto
445  * sub-system of the OpenBSD operating system. It is copyright by Vincent Rijmen, *
446  * Antoon Bosselaers and Paulo Barreto. See <a
447  * href="http://www.openbsd.org/cgi-bin/cvsweb/src/sys/crypto/rijndael.c">rijndael.c</a>
448  * for License info.
449  *
450  * @section download Getting the Files
451  *
452  * You can get the sources either from the <a
453  * href="http://sourceforge.net/projects/tinydtls/files">downloads</a> section or
454  * through git from the <a
455  * href="http://sourceforge.net/projects/tinydtls/develop">project develop page</a>.
456  *
457  * @section config Configuration
458  *
459  * Use @c configure to set up everything for a successful build. For Contiki, use the
460  * option @c --with-contiki.
461  *
462  * @section build Building
463  *
464  * After configuration, just type
465  * @code
466 make
467  * @endcode
468  * optionally followed by
469  * @code
470 make install
471  * @endcode
472  * The Contiki version is integrated with the Contiki build system, hence you do not
473  * need to invoke @c make explicitely. Just add @c tinydtls to the variable @c APPS
474  * in your @c Makefile.
475  *
476  * @addtogroup dtls_usage DTLS Usage
477  *
478  * @section dtls_server_example DTLS Server Example
479  *
480  * This section shows how to use the DTLS library functions to setup a
481  * simple secure UDP echo server. The application is responsible for the
482  * entire network communication and thus will look like a usual UDP
483  * server with socket creation and binding and a typical select-loop as
484  * shown below. The minimum configuration required for DTLS is the
485  * creation of the dtls_context_t using dtls_new_context(), and a callback
486  * for sending data. Received packets are read by the application and
487  * passed to dtls_handle_message() as shown in @ref dtls_read_cb.
488  * For any useful communication to happen, read and write call backs
489  * and a key management function should be registered as well.
490  *
491  * @code
492  dtls_context_t *the_context = NULL;
493  int fd, result;
494 
495  static dtls_handler_t cb = {
496  .write = send_to_peer,
497  .read = read_from_peer,
498  .event = NULL,
499  .get_psk_key = get_psk_key
500  };
501 
502  fd = socket(...);
503  if (fd < 0 || bind(fd, ...) < 0)
504  exit(-1);
505 
506  the_context = dtls_new_context(&fd);
507  dtls_set_handler(the_context, &cb);
508 
509  while (1) {
510  ...initialize fd_set rfds and timeout ...
511  result = select(fd+1, &rfds, NULL, 0, NULL);
512 
513  if (FD_ISSET(fd, &rfds))
514  dtls_handle_read(the_context);
515  }
516 
517  dtls_free_context(the_context);
518  * @endcode
519  *
520  * @subsection dtls_read_cb The Read Callback
521  *
522  * The DTLS library expects received raw data to be passed to
523  * dtls_handle_message(). The application is responsible for
524  * filling a session_t structure with the address data of the
525  * remote peer as illustrated by the following example:
526  *
527  * @code
528 int dtls_handle_read(struct dtls_context_t *ctx) {
529  int *fd;
530  session_t session;
531  static uint8 buf[DTLS_MAX_BUF];
532  int len;
533 
534  fd = dtls_get_app_data(ctx);
535 
536  assert(fd);
537 
538  session.size = sizeof(session.addr);
539  len = recvfrom(*fd, buf, sizeof(buf), 0, &session.addr.sa, &session.size);
540 
541  return len < 0 ? len : dtls_handle_message(ctx, &session, buf, len);
542 }
543  * @endcode
544  *
545  * Once a new DTLS session was established and DTLS ApplicationData has been
546  * received, the DTLS server invokes the read callback with the MAC-verified
547  * cleartext data as its argument. A read callback for a simple echo server
548  * could look like this:
549  * @code
550 int read_from_peer(struct dtls_context_t *ctx, session_t *session, uint8 *data, size_t len) {
551  return dtls_write(ctx, session, data, len);
552 }
553  * @endcode
554  *
555  * @subsection dtls_send_cb The Send Callback
556  *
557  * The callback function send_to_peer() is called whenever data must be
558  * sent over the network. Here, the sendto() system call is used to
559  * transmit data within the given session. The socket descriptor required
560  * by sendto() has been registered as application data when the DTLS context
561  * was created with dtls_new_context().
562  * Note that it is on the application to buffer the data when it cannot be
563  * sent at the time this callback is invoked. The following example thus
564  * is incomplete as it would have to deal with EAGAIN somehow.
565  * @code
566 int send_to_peer(struct dtls_context_t *ctx, session_t *session, uint8 *data, size_t len) {
567  int fd = *(int *)dtls_get_app_data(ctx);
568  return sendto(fd, data, len, MSG_DONTWAIT, &session->addr.sa, session->size);
569 }
570  * @endcode
571  *
572  * @subsection dtls_get_psk_info The Key Storage
573  *
574  * When a new DTLS session is created, the library must ask the application
575  * for keying material. To do so, it invokes the registered call-back function
576  * get_psk_info() with the current context and session information as parameter.
577  * When the call-back function is invoked with the parameter @p type set to
578  * @c DTLS_PSK_IDENTITY, the result parameter @p result must be filled with
579  * the psk_identity_hint in case of a server, or the actual psk_identity in
580  * case of a client. When @p type is @c DTLS_PSK_KEY, the result parameter
581  * must be filled with a key for the given identity @p id. The function must
582  * return the number of bytes written to @p result which must not exceed
583  * @p result_length.
584  * In case of an error, the function must return a negative value that
585  * corresponds to a valid error code defined in alert.h.
586  *
587  * @code
588 int get_psk_info(struct dtls_context_t *ctx UNUSED_PARAM,
589  const session_t *session UNUSED_PARAM,
590  dtls_credentials_type_t type,
591  const unsigned char *id, size_t id_len,
592  unsigned char *result, size_t result_length) {
593 
594  switch (type) {
595  case DTLS_PSK_IDENTITY:
596  if (result_length < psk_id_length) {
597  dtls_warn("cannot set psk_identity -- buffer too small\n");
598  return dtls_alert_fatal_create(DTLS_ALERT_INTERNAL_ERROR);
599  }
