235 /**
236  * Encrypts the specified \p src of given \p length, writing the
237  * result to \p buf. The cipher implementation may add more data to
238  * the result buffer such as an initialization vector or padding
239  * (e.g. for block cipers in CBC mode). The caller therefore must
240  * ensure that \p buf provides sufficient storage to hold the result.
241  * Usually this means ( 2 + \p length / blocksize ) * blocksize. The
242  * function returns a value less than zero on error or otherwise the
243  * number of bytes written.
244  *
245  * \param ctx The cipher context to use.
246  * \param src The data to encrypt.
247  * \param length The actual size of of \p src.
248  * \param buf The result buffer. \p src and \p buf must not
249  * overlap.
250  * \param aad additional data for AEAD ciphers
251  * \param aad_length actual size of @p aad
252  * \return The number of encrypted bytes on success, less than zero
253  * otherwise.
254  */
255 int dtls_encrypt(const unsigned char *src, size_t length,
256  unsigned char *buf,
257  unsigned char *nounce,
258  unsigned char *key, size_t keylen,
259  const unsigned char *aad, size_t aad_length);

261 /**
262  * Decrypts the given buffer \p src of given \p length, writing the
263  * result to \p buf. The function returns \c -1 in case of an error,
264  * or the number of bytes written. Note that for block ciphers, \p
265  * length must be a multiple of the cipher's block size. A return
266  * value between \c 0 and the actual length indicates that only \c n-1
267  * block have been processed. Unlike dtls_encrypt(), the source
268  * and destination of dtls_decrypt() may overlap.
269  *
270  * \param ctx The cipher context to use.
271  * \param src The buffer to decrypt.
272  * \param length The length of the input buffer.
273  * \param buf The result buffer.
274  * \param aad additional authentication data for AEAD ciphers
275  * \param aad_length actual size of @p aad
276  * \return Less than zero on error, the number of decrypted bytes
277  * otherwise.
278  */
